┌───────────────────────────────────────────────────────────────────────────┐
│                       DATA FLOW DIAGRAM                                   │
└───────────────────────────────────────────────────────────────────────────┘

INPUT:
  statement: str
  web_search_enabled: bool = True

      │
      ├──► STAGE 1: LLM-ONLY
      │      Input:  statement
      │      Output: verdict, confidence, reasoning
      │
      ├──► UNCERTAINTY DETECTION
      │      Input:  confidence, reasoning
      │      Output: needs_web_search (bool)
      │
      └──► STAGE 2 (conditional):
             │
             ├──► SERPER SEARCH
             │      Input:  statement (as query)
             │      Output: [SearchResult] (3 items)
             │              • title
             │              • link
             │              • snippet
             │
             ├──► FIRECRAWL SCRAPE (×3)
             │      Input:  url
             │      Output: ScrapedPage
             │              • url
             │              • markdown (3000 chars)
             │              • title
             │              • success
             │
             ├──► FORMAT EVIDENCE
             │      Input:  [SearchResult], [ScrapedPage]
             │      Output: evidence (formatted string)
             │
             └──► WEB JUDGE
                    Input:  statement, evidence
                    Output: verdict, confidence, reasoning

OUTPUT:
  Prediction(
    statement: str,
    overall_verdict: str,
    confidence: float,
    reasoning: str,
    used_web_search: bool,
    evidence: str | None
  )


┌───────────────────────────────────────────────────────────────────────────┐
│                     PERFORMANCE CHARACTERISTICS                           │
└───────────────────────────────────────────────────────────────────────────┘

PATH 1: FAST (LLM-ONLY)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  Latency:     ~2 seconds
  API Calls:   1 × LLM (Judge)
  Cost:        $ (1 LLM call)
  Accuracy:    High (for known facts)
  Triggered:   confidence >= 0.7 AND no uncertainty keywords

PATH 2: EVIDENCE (WEB-AUGMENTED)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  Latency:     ~15 seconds
  API Calls:   2 × LLM (Judge + WebJudge)
               1 × Serper Search
               3 × Firecrawl Scrape
  Cost:        $$$ (2 LLM + 4 external API calls)
  Accuracy:    Very High (evidence-based)
  Triggered:   confidence < 0.7 OR uncertainty keywords detected


┌───────────────────────────────────────────────────────────────────────────┐
│                          ERROR HANDLING                                   │
└───────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────┐
  │ _gather_web_evidence() Failure Handling │
  └─────────────────────────────────────────┘
         │
         ├──► Serper Search Fails
         │      Return: empty string ("")
         │      Effect: Falls back to LLM-only result
         │
         ├──► Firecrawl Scrape Fails (one URL)
         │      Return: "[Could not scrape content: error]"
         │      Effect: Uses other sources, continues
         │
         ├──► All Scrapes Fail
         │      Return: Evidence with snippets only
         │      Effect: Limited evidence, lower confidence
         │
         └──► Exception Raised
                Catch: try/except block
                Return: empty string ("")
                Effect: Falls back to LLM-only result

  GRACEFUL DEGRADATION: Never crashes, always returns something!
